# Basic Setup
import pandas as pd
import numpy as np

df = pd.read_excel('data_cleaning_practice_datasets.xlsx', sheet_name='Easy_Level')

# EXPLORE
df.info()                                    # Data types & nulls
df.describe()                                # Statistical summary
df.head(20)                                  # Sample rows
df.isnull().sum()                           # Missing value counts
df.duplicated().sum()                       # Duplicate count

# MISSING VALUES
df.dropna()                                 # Remove all rows with nulls
df.dropna(subset=['col1', 'col2'])         # Remove if specific columns null
df.fillna(value=0)                         # Fill nulls with value
df.fillna(method='ffill')                  # Forward fill
df['col'].fillna(df['col'].mean())         # Fill with mean

# DUPLICATES
df.drop_duplicates()                       # Remove all duplicates
df.drop_duplicates(subset=['OrderID'])    # Remove duplicates by column
df[df.duplicated(keep=False)]             # Show ALL duplicate rows

# TEXT CLEANING
df['col'] = df['col'].str.strip()         # Remove whitespace
df['col'] = df['col'].str.lower()         # Lowercase
df['col'] = df['col'].str.upper()         # Uppercase
df['col'] = df['col'].str.replace(' +', ' ', regex=True)  # Multiple spaces to one

# DATES
pd.to_datetime(df['date_col'])            # Parse to datetime
df['date_col'].dt.strftime('%Y-%m-%d')    # Format dates
df['date_col'].dt.year                     # Extract year
df['date_col'].dt.month                    # Extract month

# CATEGORICALS
df['col'] = df['col'].map({'old': 'new'}) # Map values
df['col'] = df['col'].replace(['NA', 'N/A', 'na'], np.nan)  # Multiple replaces

# VALIDATION
df[df['col'] < 0]                         # Find negative values
df[df['col'] > df['col'].quantile(0.95)]  # Find top 5% values
df[df['col'].duplicated(keep=False)]      # Find duplicate rows

# QUALITY REPORT
print(df.isnull().sum())                   # Missing values per column
print(df.dtypes)                           # Data types
print(df.describe())                       # Statistical summary
print(f"Duplicates: {df.duplicated().sum()}")
